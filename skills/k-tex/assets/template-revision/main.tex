% Extarticle class allows for ultra-small fonts (8pt).
% Landscape orientation maximizes horizontal space for multiple columns.
\documentclass[8pt, a4paper, landscape]{extarticle}

% Input and Language Encodings
\usepackage[utf8]{inputenc} % UTF-8 encoding for special characters
\usepackage[french]{babel} % French language support (adjust if writing strictly in English)
\usepackage[T1]{fontenc} % Modern font encoding
\usepackage{lmodern} % Latin Modern fonts for better rendering

% Geometry package for absolute minimum printable margins
\usepackage[margin=0.5cm, top=0.5cm, bottom=0.5cm, left=0.5cm, right=0.5cm]{geometry}

% Math packages for symbols and environments
\usepackage{amsmath, amssymb, amsthm}

% Customization of lists (itemize, enumerate)
\usepackage{enumitem}

% Section title formatting
\usepackage{titlesec}

% Colored/styled boxes for definitions, theorems, or key concepts
\usepackage{tcolorbox}

% Multicolumn layout to pack more text into a single page
\usepackage{multicol} % For the 3-column layout

% Microtype improves text justification in narrow columns, reducing awkward spacing
\usepackage{microtype} 

% Remove page numbers to save space at the bottom of the page
\pagestyle{empty}

% --- Column Configuration ---
% Space between columns
\setlength{\columnsep}{0.4cm}
% Optional: Add a vertical separator line between columns (set to 0pt to remove)
\setlength{\columnseprule}{0.1pt} 
% Remove paragraph indentation to save horizontal space
\setlength{\parindent}{0pt}

% --- List Configuration ---
% Ultra-compact lists: remove vertical spaces between items and lists
\setlist[itemize]{noitemsep, topsep=0pt, partopsep=0pt, parsep=0pt, leftmargin=1em}
\setlist[enumerate]{noitemsep, topsep=0pt, partopsep=0pt, parsep=0pt, leftmargin=1em}

% --- Section Styles (Black and White for clean printing) ---
\titleformat{\section}{\bfseries\small\uppercase}{}{0em}{}
\titleformat{\subsection}{\bfseries\footnotesize}{}{0em}{}
\titleformat{\subsubsection}{\bfseries\scriptsize}{}{0em}{}

% Hyperlinks support
\usepackage{hyperref}

% Command needed by pandoc when converting markdown lists
\providecommand{\tightlist}{\setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}

% --- Spacing Configuration ---
% Drastically reduce spacing around titles
% syntax: \titlespacing{command}{left}{before-sep}{after-sep}
\titlespacing{\section}{0pt}{3pt}{2pt}
\titlespacing{\subsection}{0pt}{2pt}{1pt}
\titlespacing{\subsubsection}{0pt}{1pt}{0pt}

% --- Custom Environments ---
% 'mybox' environment for highlighting concepts (Grayscale optimized for printing)
\newenvironment{mybox}[1]{%
    \begin{tcolorbox}[
        colback=white, % White background
        colframe=black, % Black frame
        title=\textbf{#1}, % Bold title
        fonttitle=\bfseries\scriptsize, % Title font size
        boxrule=0.6pt, % Thin border
        arc=0pt, % Sharp corners (no rounded edges)
        left=1pt, right=1pt, top=1pt, bottom=1pt, % Minimal internal padding
        sharp corners,
        after skip=2pt % Minimal space after the box
    ]
    \scriptsize % Smaller font size inside the box to pack more information
}{%
    \end{tcolorbox}
}

% Remove the date display
\date{}

\begin{document}

% Begin the 3-column layout
\begin{multicols*}{3}

% Dynamically import the content from 1.tex (usually generated by pandoc)
\input{1}
% Example of adding another section/page:
% \newpage
% \input{PC1}

\end{multicols*}

\end{document}
